' Интерфейсы


interface Positionable {
  + getPosition(): Position
}

interface Openable extends Positionable {
  + open(): void
  + close(): void
}

interface Lockable extends Openable {
  + setLocked(locked: boolean): void
  + getLocked(): boolean
}

interface Pressable extends Positionable {
  + press(): void
}

interface Walker extends Positionable {
  + walk(direction: Direction): void
}

interface Climbable extends Positionable {
  + getSteps(): int
  + getMaterial(): Material
  + getTop(): Position
  + getBottom(): Position
}

interface Climber extends Positionable {
  + climbUp(stairs: Climbable, steps: int): void
  + climbDown(stairs: Climbable, steps: int): void
}

' Enum'ы и Record'ы
enum Direction {
  LEFT
  RIGHT
  STRAIGHT
  BACKWARD
  - russian: String
  + toString(): String
}

enum Material {
  WOOD
  STONE
  - russian: String
  + toString(): String
}

enum OpenableSound {
  NONE
  SILENT
  LOUD
  - russian: String
  + toString(): String
}

enum OpenableState {
  OPENED
  CLOSED
  - russian: String
  + toString(): String
}

record Position {
  + place: String
  + area: String
}


' Классы
abstract class Passage implements Lockable {
  # isLocked: boolean
  # isOpen: boolean
  # name: String
  # soundModifier: OpenableSound
  # position: Position
  + makeSound(): void
  + open(): void
  + close(): void
  + setLocked(locked: boolean): void
  + getLocked(): boolean
  + isOpen(): boolean
  + getPosition(): Position
}

class Main {
  + main(args: String[]): void
}

class Button implements Pressable {
  - timesPressed: int
  - controlledPassage: Passage
  - location: Position
  + getPosition(): Position
  + getTimesPressed(): int
  + setControlledPassage(passage: Passage): void
  + press(): void
}

class Door extends Passage {
  - house: House
  + setLocked(state: boolean): void
  + open(): void
  + makeSound(): void
  + getHouse(): House
}

class Gate extends Passage {
  - street: Street
  + open(): void
  + makeSound(): void
  + getStreet(): Street
}

class Halfling implements Climber, Walker {
  - name: String
  - position: Position
  - elevation: int
  - canSee: HashSet<Positionable>
  + walk(direction: Direction): void
  + enter(passage: Passage): void
  + open(openable: Openable): void
  + climbUp(stairs: Climbable, steps: int): void
  + climbDown(stairs: Climbable, steps: int): void
  + findButton(wall: Wall): void
  + press(pressable: Pressable): void
  + appear(room: Room): void
  + getStatus(): String
  + setPosition(position: Position): void
  + getName(): String
  + getPosition(): Position
}

class Room implements Positionable {
  - house: House
  - position: Position
  - visitors: HashSet<Halfling>
  + addVisitor(friend: Halfling): void
  + hasVisitor(friend: Halfling): boolean
  + getPosition(): Position
  + getHouse(): House
}

class Stairs implements Climbable {
  - house: House
  - position: Position
  - top: Position
  - bottom: Position
  - steps: int
  - material: Material
  + climbUp(friend: Climber): void
  + climbDown(friend: Climber): void
  + getPosition(): Position
  + getMaterial(): Material
  + getSteps(): int
  + getTop(): Position
  + getBottom(): Position
  + getHouse(): House
}

class Wall implements Positionable {
  - house: House
  - mountedButton: Button
  - mountedDoor: Door
  - position: Position
  + mountButton(button: Button): void
  + mountDoor(door: Door): void
  + getMountedButton(): Button
  + getPosition(): Position
  + getHouse(): House
}

class House {
  - name: String
  - stairs: Stairs
  - wall: Wall
  - room: Room
  - door: Door
  + getStairs(): Stairs
  + getWall(): Wall
  + getRoom(): Room
  + getDoor(): Door
  + setStairs(stairs: Stairs): void
  + setWall(wall: Wall): void
  + setRoom(room: Room): void
  + setDoor(door: Door): void
  + getName(): String
}

class Street {
  - name: String
  - gate: Gate
  + getGate(): Gate
  + setGate(gate: Gate): void
  + getName(): String
}

}

}

}

}
